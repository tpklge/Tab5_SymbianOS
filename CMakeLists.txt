cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "1.0.0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(tab5symbian_runtime)
# Optional radio must not run a vendor constructor (and ESP_ERROR_CHECK) before
# app_main. Start it explicitly only when the user connects from the SD app.
idf_component_get_property(tab5_hosted_lib espressif__esp_hosted COMPONENT_LIB)
get_target_property(tab5_hosted_sources ${tab5_hosted_lib} SOURCES)
list(FILTER tab5_hosted_sources EXCLUDE REGEX "esp_hosted_host_init[.]c$")
set_property(TARGET ${tab5_hosted_lib} PROPERTY SOURCES "${tab5_hosted_sources}")
# Pinned ESP-Hosted 1.4.0 otherwise aborts the entire OS when the optional C6
# is missing and retries for ~1000 seconds. Preserve upstream sources/licenses;
# compile a reviewed build-local overlay with error propagation and cancellation.
idf_component_get_property(tab5_hosted_dir espressif__esp_hosted COMPONENT_DIR)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/tab5-hosted")
foreach(tab5_patch IN ITEMS api transport)
  if(tab5_patch STREQUAL "api")
    set(tab5_original "${tab5_hosted_dir}/host/api/src/esp_hosted_api.c")
    file(READ "${tab5_original}" tab5_code)
    string(REPLACE "ESP_ERROR_CHECK(esp_hosted_reconfigure());" "esp_err_t result = esp_hosted_reconfigure(); if (result != ESP_OK) return result;" tab5_code "${tab5_code}")
    string(REPLACE "ESP_ERROR_CHECK(transport_drv_reconfigure());" "esp_err_t result = transport_drv_reconfigure(); if (result != ESP_OK) return result;" tab5_code "${tab5_code}")
  else()
    set(tab5_original "${tab5_hosted_dir}/host/drivers/transport/transport_drv.c")
    file(READ "${tab5_original}" tab5_code)
    string(REPLACE "static int retry_slave_connection = 0;" "int retry_slave_connection = 0;" tab5_code "${tab5_code}")
    string(REPLACE "retry_slave_connection < MAX_RETRY_TRANSPORT_ACTIVE" "retry_slave_connection < 15" tab5_code "${tab5_code}")
    string(REPLACE "while (!is_transport_tx_ready()) {" "while (!is_transport_tx_ready()) { extern int tab5_network_cancelled(void); if (tab5_network_cancelled()) return ESP_ERR_TIMEOUT;" tab5_code "${tab5_code}")
  endif()
  set(tab5_overlay "${CMAKE_BINARY_DIR}/tab5-hosted/${tab5_patch}.c")
  file(WRITE "${tab5_overlay}" "/* Tab5 integration overlay; upstream copyright/license below. */\n${tab5_code}")
  list(REMOVE_ITEM tab5_hosted_sources "${tab5_original}")
  list(APPEND tab5_hosted_sources "${tab5_overlay}")
endforeach()
set_property(TARGET ${tab5_hosted_lib} PROPERTY SOURCES "${tab5_hosted_sources}")

# Whole-program size optimization for the SD toolkit and interpreter.
foreach(tab5_component IN ITEMS main platform html_engine lua lvgl__lvgl esp_http_client esp_http_server esp-tls esp_driver_uart esp_driver_i2c)
  idf_component_get_property(tab5_lto_lib ${tab5_component} COMPONENT_LIB)
  target_compile_options(${tab5_lto_lib} PRIVATE -flto -Os -march=rv32imafc_zicsr_zifencei_xesppie1p0)
endforeach()
target_link_options(${CMAKE_PROJECT_NAME}.elf PRIVATE -flto -march=rv32imafc_zicsr_zifencei_xesppie1p0 -Wl,-u,app_main)

# IDF 5.4.4 leaves listener/control/message sockets open if task creation fails.
# Compile an auditable build-local overlay; do not edit the installed SDK.
idf_component_get_property(tab5_httpd_lib esp_http_server COMPONENT_LIB)
idf_component_get_property(tab5_httpd_dir esp_http_server COMPONENT_DIR)
set(tab5_httpd_original "${tab5_httpd_dir}/src/httpd_main.c")
set(tab5_httpd_overlay "${CMAKE_BINARY_DIR}/tab5-httpd/httpd_main.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tools/patch_httpd.py" "${tab5_httpd_original}")
execute_process(COMMAND "${PYTHON}" "${CMAKE_SOURCE_DIR}/tools/patch_httpd.py" "${tab5_httpd_original}" "${tab5_httpd_overlay}" RESULT_VARIABLE tab5_httpd_patch_result)
if(NOT tab5_httpd_patch_result EQUAL 0)
  message(FATAL_ERROR "Cannot apply reviewed HTTPD startup cleanup")
endif()
get_target_property(tab5_httpd_sources ${tab5_httpd_lib} SOURCES)
list(FIND tab5_httpd_sources "${tab5_httpd_original}" tab5_httpd_source_index)
if(tab5_httpd_source_index EQUAL -1)
  message(FATAL_ERROR "HTTPD source layout changed; review cleanup overlay")
endif()
list(REMOVE_ITEM tab5_httpd_sources "${tab5_httpd_original}")
list(APPEND tab5_httpd_sources "${tab5_httpd_overlay}")
set_property(TARGET ${tab5_httpd_lib} PROPERTY SOURCES "${tab5_httpd_sources}")
target_include_directories(${tab5_httpd_lib} PRIVATE "${tab5_httpd_dir}/src")

# The IDF streaming read loop otherwise skips close-delimited response bodies.
idf_component_get_property(tab5_http_client_lib esp_http_client COMPONENT_LIB)
idf_component_get_property(tab5_http_client_dir esp_http_client COMPONENT_DIR)
set(tab5_http_client_original "${tab5_http_client_dir}/esp_http_client.c")
set(tab5_http_client_overlay "${CMAKE_BINARY_DIR}/tab5-http-client/esp_http_client.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tools/patch_http_client.py" "${tab5_http_client_original}")
execute_process(COMMAND "${PYTHON}" "${CMAKE_SOURCE_DIR}/tools/patch_http_client.py" "${tab5_http_client_original}" "${tab5_http_client_overlay}" RESULT_VARIABLE tab5_http_client_patch_result)
if(NOT tab5_http_client_patch_result EQUAL 0)
  message(FATAL_ERROR "Cannot apply reviewed HTTP close-delimited fix")
endif()
get_target_property(tab5_http_client_sources ${tab5_http_client_lib} SOURCES)
list(FIND tab5_http_client_sources "${tab5_http_client_original}" tab5_http_client_source_index)
if(tab5_http_client_source_index EQUAL -1)
  message(FATAL_ERROR "HTTP client source layout changed")
endif()
list(REMOVE_ITEM tab5_http_client_sources "${tab5_http_client_original}")
list(APPEND tab5_http_client_sources "${tab5_http_client_overlay}")
set_property(TARGET ${tab5_http_client_lib} PROPERTY SOURCES "${tab5_http_client_sources}")
