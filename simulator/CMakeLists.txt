cmake_minimum_required(VERSION 3.20)
project(tab5symbian_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SSL REQUIRED IMPORTED_TARGET openssl)
file(GLOB LUA_SRC "${ROOT}/components/lua/lua-5.4.9/src/*.c")
list(FILTER LUA_SRC EXCLUDE REGEX "/(lua|luac)[.]c$")
add_library(lua_host STATIC ${LUA_SRC})
target_include_directories(lua_host PUBLIC "${ROOT}/components/lua/lua-5.4.9/src")
add_library(platform_host STATIC "${ROOT}/components/platform/storage.cpp" "${ROOT}/components/platform/hash.cpp" "${ROOT}/components/platform/files.cpp" "${ROOT}/components/platform/api.cpp" "${ROOT}/components/platform/timers.cpp" "${ROOT}/components/platform/runtime.cpp" "${ROOT}/components/platform/input.cpp" "${ROOT}/components/platform/streams.cpp" "${ROOT}/components/platform/installer.cpp" "${ROOT}/components/platform/media.cpp" "${ROOT}/components/platform/video.cpp" "${ROOT}/components/platform/network.cpp" "${ROOT}/components/platform/codec_mp3.cpp" "${ROOT}/components/platform/codec_images.cpp" "${ROOT}/components/platform/codec_jpeg_write.cpp" "${ROOT}/components/platform/extensions.cpp")
target_include_directories(platform_host PUBLIC "${ROOT}/components/platform/include")
target_link_libraries(platform_host PUBLIC lua_host PkgConfig::SSL)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/managed_components/lvgl__lvgl" lvgl)
add_executable(tab5sim main.cpp "${ROOT}/components/platform/ui.cpp")
target_link_libraries(tab5sim PRIVATE platform_host lvgl PkgConfig::SDL2)
add_executable(core_tests "${ROOT}/tests/core_tests.cpp")
target_link_libraries(core_tests PRIVATE platform_host)
enable_testing()
add_test(NAME core COMMAND core_tests "${ROOT}/sdcard/SymbianOS")
add_executable(organizer_tests "${ROOT}/tests/organizer_tests.cpp")
target_link_libraries(organizer_tests PRIVATE platform_host)
add_test(NAME organizer COMMAND organizer_tests "${ROOT}/sdcard/SymbianOS")
add_executable(media_installer_tests "${ROOT}/tests/media_installer_tests.cpp")
target_link_libraries(media_installer_tests PRIVATE platform_host)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME package_fixtures COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tests/make_package_fixtures.py" "${ROOT}/build/package-fixtures")
set_tests_properties(package_fixtures PROPERTIES FIXTURES_SETUP packages)
add_test(NAME media_installer COMMAND media_installer_tests "${ROOT}/sdcard/SymbianOS" "${ROOT}/build/package-fixtures" "${ROOT}/tests/fixtures")
set_tests_properties(media_installer PROPERTIES FIXTURES_REQUIRED packages)
add_executable(media_response_tests "${ROOT}/tests/media_response_tests.cpp")
target_link_libraries(media_response_tests PRIVATE platform_host)
add_test(NAME media_response COMMAND media_response_tests "${ROOT}/sdcard/SymbianOS" "${ROOT}/tests/fixtures" "${ROOT}/build/media-validation")

add_executable(network_power_tests "${ROOT}/tests/network_power_tests.cpp")
target_link_libraries(network_power_tests PRIVATE platform_host)
add_test(NAME network_power COMMAND network_power_tests "${ROOT}/sdcard/SymbianOS")
add_executable(camera_format_tests "${ROOT}/tests/camera_format_tests.cpp")
target_include_directories(camera_format_tests PRIVATE "${ROOT}/main" "${ROOT}/managed_components/espressif__esp_video/include")
add_test(NAME camera_format COMMAND camera_format_tests)
